# Test compile options: inherit from parent + disable unused variable warnings (intentional for validation)
set(LOOM_TEST_COMPILE_OPTIONS ${LOOM_COMPILE_OPTIONS} -Wno-unused-variable -Wno-unused-but-set-variable)

# Helper function to add a test with proper compile options
function(loom_add_test name)
    cmake_parse_arguments(ARG "" "" "SOURCES;LIBRARIES;INCLUDE_DIRS" ${ARGN})
    if(NOT ARG_SOURCES)
        set(ARG_SOURCES ${name}.cpp)
    endif()
    if(NOT ARG_LIBRARIES)
        set(ARG_LIBRARIES loom::loom)
    endif()
    add_executable(${name} ${ARG_SOURCES})
    target_link_libraries(${name} PRIVATE ${ARG_LIBRARIES})
    target_compile_options(${name} PRIVATE ${LOOM_TEST_COMPILE_OPTIONS})
    if(ARG_INCLUDE_DIRS)
        target_include_directories(${name} PRIVATE ${ARG_INCLUDE_DIRS})
    endif()
    add_test(NAME ${name} COMMAND ${name})
endfunction()
