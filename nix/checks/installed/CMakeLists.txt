cmake_minimum_required(VERSION 3.30)
project(loom_installed_consumer LANGUAGES CXX)

find_package(loom CONFIG REQUIRED)
add_executable(installed-consumer main.cpp)
target_compile_features(installed-consumer PRIVATE cxx_std_26)
target_link_libraries(installed-consumer PRIVATE loom::loom)
if(TARGET loom::execution)
    target_link_libraries(installed-consumer PRIVATE loom::execution)
    target_compile_definitions(installed-consumer PRIVATE CHECK_EXECUTION)
endif()
if(TARGET loom::asio)
    target_link_libraries(installed-consumer PRIVATE loom::asio)
    target_compile_definitions(installed-consumer PRIVATE CHECK_ASIO)
endif()

enable_testing()
add_test(NAME installed-consumer COMMAND installed-consumer)
install(TARGETS installed-consumer RUNTIME DESTINATION bin)
